# ADR-026 — الدَرَجَةُ المَجّانِيَّةُ تُرفَعُ مُؤَقَّتاً، وتَقولُ ذلكَ في الكود

> **الحالَة: مُنَفَّذ — ‏2026-08-30.** قَرارُ مالِكٍ بِنَصِّه، وأَثَرُه
> **سَطرانِ** في `TierCatalog`. ولا يَنقُض ‏ADR-015: حارِسُ المَنحِ
> الذاتيِّ لَم يُمَسّ بِحَرف.

---

## ١. المُشكِلَةُ المَقيسَة — «باقَةٌ» لَيسَت باقَةً بَل سَقفُ المُنتَجِ كُلِّه

`spark` كانَت: **تَحليلٌ واحِدٌ شَهرِيّاً · ثَلاثُ تَحسيناتٍ · مَتجَرٌ
واحِدٌ · بِلا تَصديرِ دِراسَة**.

**وهذِه لَيسَت حالَةَ مَن اختارَ المَجّانِيَّةَ بَل حالَةَ كُلِّ مَن
دَخَل**، والسَبَبُ حَلقَةٌ مُغلَقَةٌ مَقيسَةٌ بِثَلاثَةِ أَطراف:

1. `StudioUser.Tier` افتِراضُها `"spark"`؛
2. والكاتِبُ الوَحيدُ لِلحَقلِ **في المُستَودَعِ كُلِّه** هُوَ
   `u.Tier = tier` في `POST /studio/billing/select`؛
3. وذاكَ السَطرُ **خَلفَ رَفضِ** `StudioTierPurchase.Refuse`، والرَفضُ
   يَشمَلُ الدَرَجاتِ **الأَربَعَ** لِأَنّ كُلَّها بِسِعرٍ > 0
   و`SelfServiceCheckoutExists = false` (‏ADR-015).

⇒ **لا سَبيلَ لِأَحَدٍ أَن يُغادِرَ `spark` إلّا بِكِتابَةٍ مُباشِرَةٍ في
قاعِدَةِ البَيانات.** فَأَرقامُها لَم تَكُن سِياسَةَ باقَةٍ بَل **سَقفَ
المُنتَجِ كُلِّه**.

**وأَثَرٌ لا يُرى مِن مِلَفِّ الدَرَجاتِ وَحدَه**:
`GET /studio/s/{id}/export.xlsx` خَلفَ `limits.AllowExport`، و`spark` =
`false` — أَي أَنّ نُقطَةَ تَصديرِ الدِراسَةِ كانَت **مَقفولَةً لِكُلِّ
مُستَأجِرٍ في المُستَودَعِ كُلِّه**.

## ٢. القَرار

### ٢-أ. تُرفَعُ الحُدود، **والأَرقامُ مَنقولَةٌ لا مَنحوتَة**

القِيَمُ الجَديدَةُ لِـ`spark` هي **بِحَرفِها** أَرقامُ `scale`:
‏**40 تَحليلاً · 200 تَحسيناً · 40 مَتجَراً · وتَصديرٌ مَفتوح**.

ومَصدَرُ تِلكَ الأَرقامِ مَكتوبٌ في المِلَفِّ نَفسِه: **تَكليفُ المالِكِ
يَومَ ‏2026-08-30**، بِحُجَّتِه — «السَقفُ يَبقى بَعيداً عَن مُستَخدِمٍ
حَقيقيّ، وقَريباً بِما يَكفي لِيُغلِقَ البابَ في وَجهِ حَلقَةٍ آلِيَّة».
فَلا رَقمَ يُخترَعُ هُنا (القاعِدَة ١٦)، ويَبقى السَقفُ **مُنتَهِياً**
فَلا يُعادُ فَتحُ الفاتورَةِ المَفتوحَةِ على مِفتاحِ نَموذَجِ لُغَةِ
المالِك.

### ٢-ب. والرَفعُ **مُؤَقَّتٌ**، وذلكَ **بَياناتٌ لا تَعليق**

```csharp
public const bool FreeTierTemporarilyRaised = true;
public static readonly TierLimits FreeTierBeforeRaise = /* 1 · 3 · 1 · false */;
```

**ولِماذا سِجِلٌّ كامِلٌ لا تَعليقٌ يَقولُ «مُؤَقَّت»**: تَعليقٌ بِلا
قيمَةٍ مَحفوظَةٍ **نِيَّةٌ لا خُطَّةُ رُجوع**. والرَفعُ مَسَّ أَربَعَةَ
حُقولٍ لا ثَلاثَة (‏`AllowExport` مِنها)، فَسِجِلٌّ واحِدٌ يَحمِلُها
كُلَّها يَمنَعُ أَن يُعادَ ثَلاثَةٌ ويُنسى رابِع.

**والعَودَةُ سَطران**: يُقلَبُ العَلَمُ، ويُعادُ صَفُّ `spark` مِن
`FreeTierBeforeRaise`.

### ٢-ج. وما لَم يُمَسّ — ونِصفُ القَرارِ هُنا

* **السِعرُ ‏99 كَما هُوَ**، وكَذلك ‏199 و‏399 و‏999. نَصُّ المالِكِ
  صَريح: **لا يُفتَرَضُ سِعر**. والرَفعُ **حُدودٌ لا تَسعير**.
* **`SelfServiceCheckoutExists = false` كَما هُوَ**، و`RefusePrice`
  بِحَرفِه — فَلا دَرَجَةَ بِسِعرٍ تُمنَحُ ذاتِيّاً (‏ADR-015 قائِمَة
  بِلا خَدش).
* **ولا `int.MaxValue`** — السَقفُ مُنتَهٍ، وفاحِصُه في
  `PaymentLeakTests` أَخضَرُ بِلا لَمس.

### ٢-د. ودَرَجَةُ السُلَّمِ الَّتي انقَلَبَت — تُصاغُ ولا تُحذَف

`The_ladder_never_goes_down` كانَ يَشتَرِط أَلّا تَقِلَّ دَرَجَةٌ عَن
أَدناها. والرَفعُ يَجعَلُ `spark` **فَوقَ** `lite` عَدَداً.

**ولا مُستَخدِمَ يَختَبِرُ هذا الانقِلاب**: لا أَحَدَ عَلى `lite` أَصلاً
— الكاتِبُ الوَحيدُ لِلحَقلِ خَلفَ رَفضٍ لا يُعبَر (§١). فَالسُلَّمُ
يَحرُسُ **سَطحَ بَيعٍ لَم يوجَد بَعد**، وهو بِعَينِه ما يَنتَظِرُه
الرَفعُ المُؤَقَّت.

فَصارَ يُقاسُ على الدَرَجاتِ **المَدفوعَةِ** ما دامَ العَلَمُ مَرفوعاً،
**ويَعودُ إلى الأَربَعِ تِلقائِيّاً** يَومَ يُقلَب — بِلا تَعديلِ حَرف.
وحَلَّ مَحَلَّ ما رُفِعَ **فاحِصٌ جَديد**: الدَرَجَةُ المَجّانِيَّةُ لا
تَتَجاوَزُ السَقفَ المُصَرَّحَ بِه. فَلا حَدَّ بَقِيَ بِلا فاحِص
(القاعِدَة ٢).

## ٣. البُرهان

| ما يُقاس | النَتيجَة |
|---|---|
| الحُدودُ مَرفوعَةٌ إلى السَقفِ القائِم | ثَلاثَةُ حُقولٍ = حُقولُ `scale` — **بِمُقارَنَةٍ لا بِرَقمٍ مَكتوبٍ في الاختِبار** |
| والتَصديرُ مَفتوحٌ لِكُلِّ دَرَجَة | صِفرُ دَرَجَةٍ بِـ`AllowExport = false` |
| والرَفعُ مُعلَّمٌ ومَحفوظَةٌ حالَتُه السابِقَة | `FreeTierTemporarilyRaised` · `FreeTierBeforeRaise` بِأَربَعَةِ حُقول |
| ولا سِعرَ تَحَرَّك | أَربَعَةُ أَسعارٍ مُثَبَّتَة |
| ولا حارِسَ ضَعُف | صِفرُ دَرَجَةٍ بِسِعرٍ تُمنَحُ ذاتِيّاً |
| ولا سَقفَ صارَ لا نِهائِيّاً | صِفرُ حَقلٍ بِـ`int.MaxValue` |
| والبَوّابَةُ ما زالَت تُغلَقُ عِندَ السَقفِ الجَديد | مُوجِبٌ وسالِبٌ عِندَ ‏40 و‏39 |

**وحارِسُ العَمى**: كُلُّ فَحصٍ يُؤَكِّدُ أَوَّلاً أَنّ الكاتالوجَ
أَربَعُ دَرَجاتٍ — فَلا يُقرَأُ «صِفرُ مُخالَفَة» مِن جَدوَلٍ فارِغ
(القاعِدَة ١٠).

## ٤. ما يَنتَظِرُ المالِك

**هذا القَرارُ مُؤَقَّتٌ بِتَعريفِه**، وما يُنهيه رَقمانِ لا يُخترَعان:

1. **سِعرُ الدَرَجَةِ المَجّانِيَّةِ وحُدودُها الدائِمَة** حينَ يَستَقِرُّ
   التَسعير؛
2. **ومُدَّةُ اشتِراكِ الدَرَجَة** — غَيرُ مَوجودَةٍ في أَيِّ وَثيقَةٍ
   اليَوم، و`PeriodDays = 30` **فَترَةُ حِصَّةٍ لا فَترَةُ اشتِراك**
   (نَصُّ `StudioTier.cs` يَقولُ ذلكَ صَراحَةً).

ويَومَ يوجَدانِ يُقلَبُ العَلَمُ ويُعادُ الصَفّ.

**مُلحَقٌ ‏2026-10-24 (تَيّارُ S14 البَند D) — رِسالَةُ لافِتَةِ الحَدِّ تَقولُ الحَدَّ ما دامَ لا يُباعُ شَيء، ولا يُعادُ حَرفٌ فَوقَ ما سَبَق**:
مَعَ `SelfServiceCheckoutExists = false` وبَعدَ `4e32559` (لا سِعرَ ولا زِرَّ لِدَرَجَةٍ لا تُمنَحُ ذاتِيّاً) بَقِيَت لافِتَةُ
`UpgradePrompt.razor` — في `/studio` و`/studio/s/{id}` — تَقولُ لِثَلاثَةِ أَسبابٍ «احصُل على … بِترقِيَة الباقَة»: دَعوَةٌ إلى فِعلٍ لا
يَكتَمِلُ مِن أَيِّ شاشَة (القاعِدَة ١٢). فَصارَت `upgrade.msg.analyze|refine|build` تَقولُ الحَدَّ وَحدَه («عَدَدُ التَحاليلِ في باقَتِكَ
مَحدودٌ لِكُلِّ شَهر.» ومِثلُها لِلتَحسيناتِ، و«عَدَدُ التَطبيقاتِ في باقَتِكَ مَحدود.»)؛ والعُنوانُ «تَتَطَلَّب هذه الميزَة تَرقيَة»
ورِسالَةُ التَصديرِ بَقِيا — يَقولانِ شَرطاً لا يَدعوانِ إلى فِعل. **مَقيس**: `8cd0d4c` تَوصيفٌ يَرسُمُ اللافِتاتِ السِتَّ عَشرَةَ (4 دَرَجاتٍ
× 4 أَسباب) بِـ`HtmlRenderer`، وبَصمَتُها بِحَجبِ نَصِّ الرِسالَةِ وَحدَه خَضراءُ قَبلُ وبَعد — الفَرقُ ثَلاثُ قِيَمٍ في `ar.json` لا غَير؛
ثُمَّ كوميتُ «fix(studio): لافِتَةُ الحَدِّ تَدعو إلى «تَرقِيَة الباقَة»…». **والعَودَة** يَومَ تُباعُ دَرَجَة (أ‑٤، والعَلَمُ `true`): القِيَمُ
الثَلاثُ القَديمَةُ بِحَرفِها — مَكتوبَةٌ في مُلَخَّصِ الصَفِّ المَقلوبِ `UpgradePromptTextsCharacterizationTests.The_quota_message_states_the_limit_and_directs_to_no_upgrade`
— ويُقلَبُ الصَفُّ في الكوميتِ نَفسِه. وصِياغَتُها حينَ لا يُباعُ شَيءٌ (تاريخُ التَجَدُّد؟ قَناةُ تَواصُل؟) ما زالَت لِلمالِك («ب» S11·١٨). (‏2026-10-24 — يُلحَقُ ولا يُعادُ حَرفٌ فَوقَه، تَيّارُ S14 البَند H: «fix(studio): لافِتَةُ الحَدِّ تَدعو إلى «تَرقِيَة الباقَة»…» = `29f15b0`؛ مَقيسٌ بِـ`git show -s`.)

**مُلحَقٌ ‏2026-10-25 (تَدقيقُ الإطلاق، بَندُ المُنَسِّقِ في تَيّارِ W8-3) — رَفضُ وَكيلِ الاستوديو عِندَ حَدِّ المَتاجِرِ يَقولُ الحَدَّ وَحدَه، ولا يُعادُ حَرفٌ فَوقَ ما سَبَق**:
`create_tenant` حينَ تَرُدُّه `CheckBuildAsync` كانَ يَقولُ «بَلَغتَ حَدَّ التَطبيقاتِ في باقَتِك (N مِن M). اِحذِف تَطبيقاً أَو اِرفَع باقَتَك مِن /studio/billing.»
— والبابانِ لا يُفتَحان، مَقيسَين: `StudioUser.StoresBuilt` لا يَكتُبُه إلّا `StoresBuilt++` (فَالحَذفُ لا يُحَرِّرُ حِصَّة، ولا بابَ حَذفٍ تَحتَ
`/studio/apps/{slug}` أَصلاً — صِفرٌ مِن 40 مَساراً)، ولا دَرَجَةَ `IsSelfGrantable` (صِفرٌ مِن 4) — وهو ما قاسَه «أ» S14·٤ في الطَريقِ ولَم يَمَسَّه.
فَسَقَطَت الجُملَةُ الثانِيَةُ وَحدَها، بِحَرفِها وبِلا حَرفٍ جَديد (`studio.agent.err_store_limit` في `ar.json`): «بَلَغتَ حَدَّ التَطبيقاتِ في باقَتِك ({0} مِن {1}).»
**مَقيس**: `f555fdf` تَوصيفٌ («Today_») ثُمَّ كوميتُ الصِياغَة؛ والبُرهانُ الحَيُّ `LiveAgentStoreLimitRefusalProofTests` عَلى Postgres مَحَلِّيّ ‏173 ← 84 بايتاً
(الفَرقُ ‏89 = الجُملَةُ ومَسافَتُها). **والعَودَة**: صَفُّ `AgentStoreLimitRefusalTests.The_store_limit_refusal_states_the_limit_and_sends_to_no_door_that_does_not_open`
يَقيسُ المُقَدِّماتِ الثَلاثَ بِأَعيانِها ويُحمِرُّ يَومَ يُنقِصُ بابٌ العَدّادَ أَو تُباعُ دَرَجَةٌ أَو يولَدُ بابُ حَذف — والجُملَةُ حينَئِذٍ صِياغَةُ المالِك
(«ب» S11·١٨ و·٢١)، ونُقصانُ العَدّادِ عِندَ الحَذفِ خِيارٌ لَه لا لِلوَكيل.

**مُلحَقٌ ‏2026-10-29 (قَرارُ المالِك د15 (+د46) «ب») — صِياغَةُ «حينَ لا يُباعُ شَيء» كَتَبَها المالِك: جُملَةٌ ثانِيَةٌ تُعطي خُطوَةً مُتاحَة، ولا يُعادُ حَرفٌ فَوقَ ما سَبَق**:
ما تَرَكَه المُلحَقانِ أَعلاهُ لِلمالِكِ («تاريخُ التَجَدُّد؟ قَناةُ تَواصُل؟» — «ب» S11·١٨) صارَ: (١) لِلتَحاليلِ والتَحسيناتِ «تَتَجَدَّدُ حِصَّتُكَ في {التاريخ}.»
والتاريخُ بِدايَةُ الفَترَةِ + ‏30 يَوماً (`StudioLimitNextStep.RenewsAt` — `null` إن انقَضَت الفَترَة، فَلا تاريخَ في الماضي)، تَحسِبُه `StudioHome` و`StudioStudy`
وتُنَسِّقانِه بِساعَةِ الطَلَب (`Culture.FormatTime`) وتُمَرِّرانِه وَسيطاً `RenewsOn`؛ (٢) لِكُلِّ حَدٍّ — ومَعَه رَفضُ وَكيلِ الاستوديو عِندَ حَدِّ التَطبيقات —
«لِرَفعِ الحَدِّ تَواصَل مَعَنا عَلى {بَريدُ التَواصُل}.» **ولا تُرسَمُ ما دامَ `platform.doc.contact_email` نائِباً** (الآلِيَّةُ نَفسُها في صَفحَتَي الأَسعارِ والتَواصُل —
`StudioLimitNextStep.Contact`، شَرطٌ واحِدٌ لِلمُستَهلِكَين)؛ (٣) وعُنوانُ الميزَةِ ورِسالَةُ التَصدير «هذِه الميزَةُ غَيرُ مَشمولَةٍ في باقَتِك» و«تَصديرُ الدِراسَةِ
غَيرُ مَشمولٍ في باقَتِك.» تَتبَعُهُما جُملَةُ التَواصُل. **والصِياغَةُ تُختارُ بِالشَرطِ نَفسِه الَّذي يَرسُمُ الزِرَّ والسِعر** (`Offer`، أَي `StudioTierPurchase.IsSelfGrantable`
لِلدَرَجَةِ التالِيَة): يَومَ تُباعُ دَرَجَةٌ تَعودُ صِياغَةُ «التَرقِيَة» وَحدَها — والمِفتاحانِ القَديمانِ باقِيانِ في القامُوسِ بِحَرفِهِما. وقالَبُ رَفضِ الوَكيلِ لَم
يُمَسّ؛ الجُملَةُ تُلحَقُ بِه في الكود. **مَقيس**: `4effde3` تَوصيفُ «اليَوم» («وَسائِطُ اللافِتَة: CurrentTier, Reason · لافِتاتٌ بِخُطوَةٍ تالِيَة: 0/16»)، ثُمَّ كوميتُ
التَبديل؛ وبَصمَةُ اللافِتاتِ بِحَجبِ الرِسالَةِ أُعيدَ تَثبيتُها (‏`b1c517f6d42a11eb` ← `6f2f44c48d8a40c5`) والصَفُّ نَفسُه يُبَرهِنُ أَنّ الفَرقَ عُنوانُ لافِتَةِ التَصديرِ
وَحدَه: بِعُنوانِه القَديمِ تَعودُ البَصمَةُ القَديمَةُ حَرفاً. **وعَلى المالِك**: مَلءُ `platform.doc.contact_email` (`DEPLOY.md` §٢ الخُطوَة ٧) لِتَظهَرَ جُملَةُ التَواصُل.

## مُلحَق د16 — مَن عَلِقَ بِدَرَجَةٍ أَضيَقَ مِن المَجّانِيَّةِ يُعادُ إلَيها (‏2026-10-29، قَرارُ المالِك د16 «أ»)

> يُلحَقُ ولا يُعادُ حَرفٌ فَوقَه.

مَن خُزِّنَت لَه `lite` (‏3 · 10 · 3) أَو `growth` (‏10 · 50 · 10) قَبلَ إغلاقِ `/studio/billing/select` يَومَ ‏2026-08-30 صارَ بِهذا الرَفعِ أَسوَأَ
حالاً مِن المُستَخدِمِ الجَديد (‏40 · 200 · 40)، ولا رُجوعَ لَه («ج» S11·١٩). **أَفضَلُ المُمارَساتِ أَلّا يَكونَ العَميلُ الأَقدَمُ أَسوَأَ حالاً**،
وهُم عُملاءُ وَسايِل نَفسِها. فَـ`scripts/repair-studio-tier.cs` يَعرِضُ العَدَدَ ومَن سَيَتَغَيَّرُ (بِالمُعَرِّفاتِ والدَرَجاتِ وَحدَها) ثُمَّ يُطَبِّقُ
بِـ`--apply --confirm N`، ويَكتُبُ لِكُلِّ مُستَخدِمٍ سَطرَ تَدقيقٍ `billing.tier.repair_to_free` بِدَرَجَتِه السابِقَة (`before`) وأَمرَ استِرجاعٍ لا بَياناتَ
شَخصِيَّةً فيه. **والقَرارُ دالَّةٌ نَقِيَّة** (`StudioTierRepair.IsStuck`): دَرَجَةٌ مَعروفَةٌ تَتَفَوَّقُ عَلَيها المَجّانِيَّةُ اليَومَ بِالأَرقام (`TierCatalog.Exceeds`) —
فَـ`scale` لا تُمَسّ، ويَومَ يُنهى الرَفعُ لا يَعودُ مُرَشَّحٌ واحِد.

**ويَومَ يُنهى الرَفعُ المُؤَقَّتُ — سَطرُ قَرارٍ لا كود (القاعِدَة ١)**: يُعادُ كُلُّ مَن أُعيدَ هُنا إلى **الأَوسَعِ** بَينَ دَرَجَتِه السابِقَةِ (مِن سَطرِ
التَدقيق) والمَجّانِيَّةِ يَومَئِذٍ، لا إلى أَضيَقَ مِنها. لا يُبنى ذلك الآن: لا مُستَهلِكَ لِلدَرَجَةِ السابِقَةِ قَبلَ قَلبِ `FreeTierTemporarilyRaised`،
والحَقيقَةُ مَحفوظَةٌ في التَدقيقِ لا في حَقلٍ جَديد. **والتَشغيلُ عَلى الإنتاجِ فِعلُ المالِك** بَعدَ العَدِّ عَلى فَرعٍ مَنسوخ («ج» د16).


## مُلحَق د14 — البَيعُ عِندَ الإطلاقِ تَرتيبٌ إدارِيّ، وأَسعارُ الدَرَجاتِ تَبقى غَيرَ مَعروضَة (‏2026-10-29، قَرارُ المالِك د14 «أ» — يُلحَقُ ولا يُعادُ حَرفٌ فَوقَ ما سَبَق)

1. **نَموذَجُ البَيعِ عِندَ الإطلاق — تَرتيبٌ إدارِيّ** (نَصُّ القَرار): يَتَّفِقُ المالِكُ مَعَ العَميل، ويُسَجِّلُ سِعرَ اشتِراكِ مَتجَرِه ومُدَّتَه مِن لَوحَةِ المُشرِف
   (`/admin/tenants/{slug}/plan` و`…/plan/save`)، ويُحَصَّلُ بِالتَحويلِ أَو بِرابِطِ PayPal القائِم (`…/plan/paypal-link`)، ولَه أَن يَمنَحَ مَن يَشاءُ بِسِعرِ صِفر.
   والسَبَب: «أفضل الممارسات لفريق فرديّ قبل إثبات الطلب أن يبدأ بالبيع اليدويّ، ولا يبني الدفع الذاتيّ إلا بعد أن يتكرّر الطلب». **ولا بِناء**:
   الأَبوابُ قائِمَة، وحُدودُ الاستوديو المَجّانِيَّةُ مَرفوعَةٌ إلى حُدودِ أَعلى دَرَجَةٍ (`FreeTierTemporarilyRaised = true`، ‏`StudioTier.cs:73`).
2. **أَسعارُ الدَرَجاتِ الأَربَعِ ودَورِيَّتُها — تَبقى غَيرَ مَعروضَةٍ حَتّى يَكتُبَ المالِكُ أَرقامَه** (القاعِدَة ١٦): ما في الكودِ (‏`spark` ‏99، و`lite` ‏199، و`growth` ‏399، و`scale` ‏999)
   لا يَراه أَحَد — `SelfServiceCheckoutExists = false` (‏`StudioTier.cs:227`). ويَومَ تُكتَبُ الأَسعارُ تُكتَبُ في `docs/` ثُمَّ في الكود،
   وبُرهانُها اختِباراتُ أَسعارِ الاستوديو ولَقطَةُ صَفحَةِ الفَوتَرَة. والسُؤالُ باقٍ لِلمالِكِ في سِجِلِّه، ولا يَلزَمُ قَبلَ إنهاءِ الرَفعِ أَو فَتحِ البَيعِ الذاتِيّ.
3. **وعَلى المالِك** (‏«ج»): لِكُلِّ عَميلٍ يَتَّفِقُ مَعَه يُسَجِّلُ سِعرَ مَتجَرِه ومُدَّتَه مِن صَفحَةِ باقَةِ المَتجَر؛ ولا يَفتَحُ سَماحاً ولا رابِطَ PayPal لِمَتجَرٍ سِعرُه صِفر.
